**Goal:** Give `FitResult` a role-structured quality-control surface — a scalar pass/fail gate (`qc_summary`) plus drill-down plots (convergence of shape params, area forest, warp/noise forest, R̂-vs-ESS overview) — that scales to the model's many parameters instead of dumping one subplot per scalar.

**Architecture:** A new pure module `chromhandler/fitting/qc.py` holds functions that take an `arviz` `DataTree` (the fit's `idata`) and return a `matplotlib.Figure` or a metric dict — no `FitResult` dependency, so they're unit-testable on a synthetic `idata` with no MCMC. `FitResult` gains thin wrapper methods. Plots are chosen by parameter *role*: few interpretable params (shape) get rank plots; many params (areas, warp, noise) get forest plots; everything gets a single R̂/ESS overview. Fixes item 13 (default var-names include `*_warped`) along the way.

**Tech Stack:** arviz 1.1.0 (`plot_rank`, `plot_forest`, `plot_energy`, `summary`; `PlotCollection.viz["figure"].item()` → Figure), matplotlib, numpy.

## Background (verified API facts)

- `arviz.plot_rank(dt, *, var_names=...)`, `arviz.plot_forest(dt, *, var_names=...)` → `PlotCollection`; get the figure with `pc.viz["figure"].item()` (returns a `matplotlib.figure.Figure`).
- `arviz.summary(idata, var_names=...)` → DataFrame indexed by scalar param label (`"area[0, 0]"`, `"mu[0]"`) with columns incl. `r_hat`, `ess_bulk`.
- BFMI per chain `= mean(diff(energy)²) / var(energy)` over draws, from `idata.sample_stats["energy"]` (shape `[chain, draw]`).
- Current model posterior groups: `mu`,`width`,`skew` `[n_peak]`; `area` `[n_trace,n_peak]`; `time_shift`,`time_stretch`,`noise`,`baseline_intercept`,`baseline_slope` `[n_trace]`; `mu_warped`,`width_warped` `[n_trace,n_peak]` (derived); plus `*_raw` sample sites.

```python
"""Quality-control views over a fit's InferenceData.

Pure functions: each takes an arviz DataTree (`idata`) and returns a
matplotlib Figure or a metric dict. No FitResult/dataset dependency, so
they are unit-testable on a synthetic idata. Plots are chosen by parameter
ROLE so the view scales to the model's many parameters:
  - shape (mu/width/skew): few, interpretable      -> rank plots
  - area[trace, peak]:      many, the estimands      -> forest
  - warp + noise (per trace): nuisance               -> forest
  - everything:             R-hat / ESS overview     -> scatter + scalar gate
"""
from __future__ import annotations

from typing import TYPE_CHECKING, Any

import arviz
import numpy as np

if TYPE_CHECKING:
    import matplotlib.figure


def _figure(pc: Any) -> "matplotlib.figure.Figure":
    """Extract the matplotlib Figure from an arviz 1.x PlotCollection."""
    return pc.viz["figure"].item()


def _present(idata: arviz.InferenceData, names: tuple[str, ...]) -> list[str]:
    """Subset of ``names`` actually present in the posterior."""
    data_vars = idata.posterior.data_vars  # type: ignore[attr-defined]
    return [n for n in names if n in data_vars]
```

    idata: arviz.InferenceData,

```python
# Map a posterior variable name to its QC role group.
_GROUP = {
    "mu": "shape", "width": "shape", "skew": "shape",
    "area": "area",
    "time_shift": "warp", "time_stretch": "warp",
    "noise": "noise",
    "baseline_intercept": "baseline", "baseline_slope": "baseline",
}


def _var_of(label: str) -> str:
    """'area[0, 0]' -> 'area'."""
    return label.split("[", 1)[0]


def _bfmi(idata: arviz.InferenceData) -> np.ndarray:
    """Per-chain BFMI from sample_stats.energy (empty if absent)."""
    if not (hasattr(idata, "sample_stats") and "energy" in idata.sample_stats):  # type: ignore[attr-defined]
        return np.array([])
    e = np.asarray(idata.sample_stats["energy"])  # type: ignore[attr-defined]  [chain, draw]
    num = np.mean(np.diff(e, axis=1) ** 2, axis=1)
    den = np.var(e, axis=1)
    return num / np.where(den > 0, den, np.nan)


def qc_summary(idata: arviz.InferenceData) -> dict[str, Any]:
    """Scalar QC gate + per-role worst R-hat / min ESS.

    Returns a dict:
        fit_healthy: bool, n_divergent: int, bfmi_min: float,
        rhat_max: float, ess_min: float,
        groups: {group: {rhat_max, ess_min}} for shape/area/warp/noise/baseline.
    Healthy iff r_hat <= 1.01 everywhere, ess_bulk > 400 everywhere,
    no divergences, and bfmi_min >= 0.3.
    """
    names = [
        str(n) for n in idata.posterior.data_vars  # type: ignore[attr-defined]
        if not (str(n).endswith("_raw") or str(n).endswith("_warped"))
    ]
    summ = arviz.summary(idata, var_names=names)
    rhat = summ["r_hat"].astype(float)
    ess = summ["ess_bulk"].astype(float)

    groups: dict[str, dict[str, float]] = {}
    for label in summ.index:
        grp = _GROUP.get(_var_of(str(label)), "other")
        d = groups.setdefault(grp, {"rhat_max": -np.inf, "ess_min": np.inf})
        d["rhat_max"] = max(d["rhat_max"], float(rhat[label]))
        d["ess_min"] = min(d["ess_min"], float(ess[label]))

    n_div = 0
    if hasattr(idata, "sample_stats") and "diverging" in idata.sample_stats:  # type: ignore[attr-defined]
        n_div = int(np.asarray(idata.sample_stats["diverging"]).sum())  # type: ignore[attr-defined]
    bfmi = _bfmi(idata)
    bfmi_min = float(np.nanmin(bfmi)) if bfmi.size else float("nan")

    rhat_max = float(rhat.max())
    ess_min = float(ess.min())
    rhat_ok = bool(np.isnan(rhat_max) or rhat_max <= 1.01)
    bfmi_ok = bool(np.isnan(bfmi_min) or bfmi_min >= 0.3)
    fit_healthy = bool(rhat_ok and ess_min > 400 and n_div == 0 and bfmi_ok)

    return {
        "fit_healthy": fit_healthy,
        "n_divergent": n_div,
        "bfmi_min": bfmi_min,
        "rhat_max": rhat_max,
        "ess_min": ess_min,
        "groups": groups,
    }


def plot_qc_overview(idata: arviz.InferenceData) -> "matplotlib.figure.Figure":
    """One-panel R-hat-vs-ESS scatter over all params, coloured by role group.

    Catches any single problematic parameter among hundreds without one
    subplot per scalar. Reference lines: r_hat = 1.01, ess = 400.
    """
    import matplotlib.pyplot as plt

    names = [
        str(n) for n in idata.posterior.data_vars  # type: ignore[attr-defined]
        if not (str(n).endswith("_raw") or str(n).endswith("_warped"))
    ]
    summ = arviz.summary(idata, var_names=names)
    groups = [_GROUP.get(_var_of(str(i)), "other") for i in summ.index]
    fig, ax = plt.subplots(figsize=(7.0, 4.5))
    for grp in sorted(set(groups)):
        m = [g == grp for g in groups]
        ax.scatter(
            summ["ess_bulk"][m].astype(float),
            summ["r_hat"][m].astype(float),
            s=18, alpha=0.7, label=grp,
        )
    ax.axhline(1.01, color="r", lw=0.8, ls="--")
    ax.axvline(400, color="r", lw=0.8, ls="--")
    ax.set_xlabel("ESS (bulk)")
    ax.set_ylabel("R-hat")
    ax.set_title("Convergence overview (target: r_hat < 1.01, ESS > 400)")
    ax.legend(fontsize=8)
    fig.tight_layout()
    return fig
```
